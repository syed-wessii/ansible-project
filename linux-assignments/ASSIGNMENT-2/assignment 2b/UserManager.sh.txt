#!/bin/bash

# Ensure the script is run with root privileges
if [ "$EUID" -ne 0 ]; then
  echo "Please run this script with sudo or as root."
  exit 1
fi

COMMAND=$1
TARGET=$2
PARAM=$3

if [ "$COMMAND" == "addTeam" ]; then
    groupadd "$TARGET"
    echo "Team '$TARGET' created successfully."

elif [ "$COMMAND" == "addUser" ]; then
    # Create the global 'ninja' group if it doesn't exist
    getent group ninja >/dev/null || groupadd ninja

    # Check if the team group exists before adding the user
    if ! getent group "$PARAM" >/dev/null; then
        echo "Error: Team '$PARAM' does not exist. Please create it first."
        exit 1
    fi

    # Create the user:
    # -m creates the home directory
    # -g sets the primary group (their team)
    # -G adds them to the supplementary 'ninja' group
    # -s sets their default shell
    useradd -m -g "$PARAM" -G ninja -s /bin/bash "$TARGET"

    # Set home directory permissions:
    # 7 (User): Read, Write, Execute
    # 5 (Group): Read, Execute
    # 1 (Others): Execute only
    chmod 751 "/home/$TARGET"

    # Create shared directories
    mkdir -p "/home/$TARGET/team"
    mkdir -p "/home/$TARGET/ninja"

    # Set 'team' directory permissions (Full access to user and team members)
    chown "$TARGET":"$PARAM" "/home/$TARGET/team"
    chmod 770 "/home/$TARGET/team"

    # Set 'ninja' directory permissions (Full access to user and all ninjas)
    chown "$TARGET":ninja "/home/$TARGET/ninja"
    chmod 770 "/home/$TARGET/ninja"

    echo "User '$TARGET' added to team '$PARAM' with proper directory structures."

elif [ "$COMMAND" == "delTeam" ]; then
    groupdel "$TARGET"
    echo "Team '$TARGET' deleted."

elif [ "$COMMAND" == "delUser" ]; then
    # -r removes the user's home directory as well
    userdel -r "$TARGET"
    echo "User '$TARGET' and their home directory deleted."

elif [ "$COMMAND" == "changePasswd" ]; then
    passwd "$TARGET"

elif [ "$COMMAND" == "changeShell" ]; then
    usermod -s "$PARAM" "$TARGET"
    echo "Shell for '$TARGET' changed to '$PARAM'."

elif [ "$COMMAND" == "ls" ]; then
    if [ "$TARGET" == "User" ]; then
        echo "--- System Users (UID >= 1000) ---"
        # Parses /etc/passwd for standard users
        awk -F: '($3>=1000 && $1!="nobody") {print $1}' /etc/passwd
    elif [ "$TARGET" == "Team" ]; then
        echo "--- Custom Teams (GID >= 1000) ---"
        # Parses /etc/group for standard groups
        awk -F: '($3>=1000 && $1!="nogroup") {print $1}' /etc/group
    else
        echo "Usage: ./UserManager.sh ls [User|Team]"
    fi

else
    echo "Usage: ./UserManager.sh [addTeam|addUser|delTeam|delUser|changePasswd|changeShell|ls] ..."
fi
