#!/bin/bash

DB="$HOME/.otssh.db"
touch "$DB"

########################################
# LIST
########################################
if [ "$1" = "ls" ]; then
    if [ "$2" = "-d" ]; then
        while IFS="|" read -r name host user port key
        do
            [ -z "$name" ] && continue

            cmd="ssh"

            [ -n "$key" ] && cmd="$cmd -i $key"
            [ "$port" != "22" ] && cmd="$cmd -p $port"

            cmd="$cmd $user@$host"

            echo "$name: $cmd"
        done < "$DB"
    else
        cut -d'|' -f1 "$DB"
    fi
    exit
fi

########################################
# DELETE
########################################
if [ "$1" = "rm" ]; then
    if [ -z "$2" ]; then
        echo "[ERROR]: Please provide a server name. Example: otssh rm myserver"
        exit 1
    fi

    if ! grep -q "^$2|" "$DB"; then
        echo "[ERROR]: Server '$2' not found."
        exit 1
    fi

    tmp=$(mktemp)

    grep -v "^$2|" "$DB" > "$tmp"

    mv "$tmp" "$DB"

    echo "$2 deleted."
    exit
fi

########################################
# CONNECT
########################################
if [[ "$1" != -* && "$1" != "ls" && "$1" != "rm" && -n "$1" ]]; then

    line=$(grep "^$1|" "$DB")

    if [ -z "$line" ]; then
        echo "[ERROR]: Server information is not available, please add server first"
        exit 1
    fi

    IFS="|" read -r name host user port key <<< "$line"

    echo "Connecting to $name"
    echo "Host : $host"
    echo "User : $user"
    echo "Port : $port"
    echo "Key  : $key"
    echo

    cmd="ssh"

    [ -n "$key" ] && cmd="$cmd -i $key"
    [ "$port" != "22" ] && cmd="$cmd -p $port"

    echo "Command Running......"
    echo "$cmd $user@$host"
    echo

    echo "..............Connecting............."

    eval "$cmd $user@$host"

    exit
fi

########################################
# ADD / UPDATE
########################################
ACTION=""

while getopts ":aUn:h:u:p:i:" opt
do
    case $opt in
        a) ACTION="ADD" ;;
        U) ACTION="UPDATE" ;;
        n) NAME=$OPTARG ;;
        h) HOST=$OPTARG ;;
        u) USERNAME=$OPTARG ;;
        p) PORT=$OPTARG ;;
        i) KEY=$OPTARG ;;

        *)
            echo "[ERROR]: Invalid option."
            echo "Usage: otssh -a|-U -n NAME -h HOST -u USER [-p PORT] [-i KEY]"
            exit 1
            ;;
    esac
done

PORT=${PORT:-22}

########################################
# VALIDATION
########################################

if [ -z "$ACTION" ]; then
    echo "[ERROR]: You must specify -a (add) or -U (update)."
    exit 1
fi

if [ -z "$NAME" ]; then
    echo "[ERROR]: -n NAME is required."
    exit 1
fi

if [ -z "$HOST" ]; then
    echo "[ERROR]: -h HOST is required."
    exit 1
fi

if [ -z "$USERNAME" ]; then
    echo "[ERROR]: -u USERNAME is required."
    exit 1
fi

########################################
# ADD
########################################

if [ "$ACTION" = "ADD" ]; then

    if grep -q "^$NAME|" "$DB"; then
        echo "Server already exists."
        exit 1
    fi

    echo "$NAME|$HOST|$USERNAME|$PORT|$KEY" >> "$DB"

    echo "Server added successfully."

    exit
fi

########################################
# UPDATE
########################################

if [ "$ACTION" = "UPDATE" ]; then

    if ! grep -q "^$NAME|" "$DB"; then
        echo "[ERROR]: Server '$NAME' not found, use -a to add it first."
        exit 1
    fi

    tmp=$(mktemp)

    grep -v "^$NAME|" "$DB" > "$tmp"

    echo "$NAME|$HOST|$USERNAME|$PORT|$KEY" >> "$tmp"

    mv "$tmp" "$DB"

    echo "Server updated successfully."

    exit
fi
