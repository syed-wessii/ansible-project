#!/bin/bash

# Define the hidden database file
DB_FILE="$HOME/.process_manager.db"

# Ensure the database file exists
if [[ ! -f "$DB_FILE" ]]; then
    touch "$DB_FILE"
fi

# --- Variables ---
OPERATION=""
SCRIPT_PATH=""
ALIAS=""
PRIORITY_LEVEL=""

# --- The Router: Parse the flags ---
while [[ $# -gt 0 ]]; do
    case "$1" in
        -o) OPERATION="$2"; shift 2 ;;
        -s) SCRIPT_PATH="$2"; shift 2 ;;
        -a) ALIAS="$2"; shift 2 ;;
        -p) PRIORITY_LEVEL="$2"; shift 2 ;;
        *) shift ;;
    esac
done

# --- Functions (The Engine) ---

update_db_pid() {
    local target_alias=$1
    local new_pid=$2
    TMP_FILE=$(mktemp)

    # Read line by line using the IFS pipe trick
    while IFS="|" read -r db_alias db_path db_pid; do
        if [[ "$db_alias" == "$target_alias" ]]; then
            echo "${db_alias}|${db_path}|${new_pid}" >> "$TMP_FILE"
        else
            echo "${db_alias}|${db_path}|${db_pid}" >> "$TMP_FILE"
        fi
    done < "$DB_FILE"
    mv "$TMP_FILE" "$DB_FILE"
}

register_process() {
    if grep -q "^${ALIAS}|" "$DB_FILE"; then
        echo "[ERROR]: Alias '$ALIAS' is already registered."
        exit 1
    fi
    # Store format: alias|path|pid (PID is empty initially)
    echo "${ALIAS}|${SCRIPT_PATH}|" >> "$DB_FILE"
    echo "Success: Process '$ALIAS' registered."
}

start_process() {
    # Extract the script path for the given alias
    TARGET_PATH=$(awk -F'|' -v a="$ALIAS" '$1==a {print $2}' "$DB_FILE")

    if [[ -z "$TARGET_PATH" ]]; then
        echo "[ERROR]: Alias '$ALIAS' not found."
        exit 1
    fi

    echo "Starting $ALIAS as a background daemon..."
    # Run the script in the background, ignoring hangups
    nohup bash "$TARGET_PATH" > /dev/null 2>&1 &

    # $! grabs the Process ID (PID) of the last background command run
    NEW_PID=$!

    update_db_pid "$ALIAS" "$NEW_PID"
    echo "Success: '$ALIAS' started with PID: $NEW_PID"
}

status_process() {
    TARGET_PID=$(awk -F'|' -v a="$ALIAS" '$1==a {print $3}' "$DB_FILE")

    if [[ -z "$TARGET_PID" ]]; then
        echo "Service '$ALIAS' is NOT running (No PID found)."
        return
    fi

    # Check if the process exists silently
    if ps -p "$TARGET_PID" > /dev/null; then
        echo "Service '$ALIAS' is RUNNING (PID: $TARGET_PID)."
    else
        echo "Service '$ALIAS' is STOPPED."
    fi
}

kill_process() {
    TARGET_PID=$(awk -F'|' -v a="$ALIAS" '$1==a {print $3}' "$DB_FILE")

    if [[ -n "$TARGET_PID" ]] && ps -p "$TARGET_PID" > /dev/null; then
        kill -9 "$TARGET_PID"
        echo "Success: Sent termination signal to '$ALIAS' (PID: $TARGET_PID)."
        # Clear the PID from the database
        update_db_pid "$ALIAS" ""
    else
        echo "[ERROR]: '$ALIAS' is not currently running."
    fi
}

change_priority() {
    TARGET_PID=$(awk -F'|' -v a="$ALIAS" '$1==a {print $3}' "$DB_FILE")

    if [[ -z "$TARGET_PID" ]] || ! ps -p "$TARGET_PID" > /dev/null; then
        echo "[ERROR]: Cannot change priority. '$ALIAS' is not running."
        exit 1
    fi

    # Map high/med/low to nice values (-20 to +19)
    case "$PRIORITY_LEVEL" in
        high) NICE_VAL="-10" ;;
        med)  NICE_VAL="0" ;;
        low)  NICE_VAL="10" ;;
        *) echo "[ERROR]: Priority must be high, med, or low."; exit 1 ;;
    esac

    echo "Changing priority of '$ALIAS' to $PRIORITY_LEVEL (Nice value: $NICE_VAL)..."
    renice -n "$NICE_VAL" -p "$TARGET_PID"
}

list_aliases() {
    # Print only the first column (the alias)
    awk -F'|' '{print $1}' "$DB_FILE"
}

top_processes() {
    echo "Alias, PID, State, Priority, Script"
    echo "---------------------------------------------------"

    while IFS="|" read -r db_alias db_path db_pid; do
        # If an alias was provided, skip the rest
        if [[ -n "$ALIAS" ]] && [[ "$ALIAS" != "$db_alias" ]]; then
            continue
        fi

        if [[ -n "$db_pid" ]] && ps -p "$db_pid" > /dev/null; then
            # Extract stat and nice values dynamically without headers
            STATE=$(ps -p "$db_pid" -o stat= | tr -d ' ')
            PRIO=$(ps -p "$db_pid" -o ni= | tr -d ' ')
        else
            db_pid="-"
            STATE="STOPPED"
            PRIO="-"
        fi

        echo "$db_alias, $db_pid, $STATE, $PRIO, $db_path"
    done < "$DB_FILE"
}

# --- Action Execution ---
case "$OPERATION" in
    register) register_process ;;
    start)    start_process ;;
    status)   status_process ;;
    kill)     kill_process ;;
    priority) change_priority ;;
    list)     list_aliases ;;
    top)      top_processes ;;
    *)        echo "[ERROR]: Invalid operation. Use register, start, status, kill, priority, list, or top." ;;
esac
