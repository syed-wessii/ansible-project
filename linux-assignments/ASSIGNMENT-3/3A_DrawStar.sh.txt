#!/bin/bash


size=5

for (( i = 1; i <= size; i++ )); do

    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done

    for (( j = 1; j <= i; j++ )); do
        echo -n "*"
    done
    echo
done

echo ""

for (( i = 1; i <= size; i++ )); do
    for (( j = 1; j <= i; j++ )); do
        echo -n "*"
    done
    echo
done

echo ""


echo "centered pyramid"
for (( i = 1; i <= size; i++ )); do
    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done
    for (( j = 1; j <= 2*i - 1; j++ )); do
        echo -n "*"
    done
    echo
done

echo ""

for (( i = size; i >= 1; i-- )); do
    for (( j = 1; j <= i; j++ )); do
        echo -n "*"
    done
    echo
done

echo ""

for (( i = size; i >= 1; i-- )); do
    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done

    for (( j = 1; j <= i; j++ )); do
        echo -n "* "
    done

    echo
done

echo ""

for (( i = size; i >= 1; i-- )); do
    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done

    for (( j = 1; j <= 2*i - 1; j++ )); do
        echo -n "*"
    done

    echo
done

echo ""

for (( i = 1; i <= size; i++ )); do
    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done
    for (( j = 1; j <= 2*i - 1; j++ )); do
        echo -n "*"
    done
    echo
done

for (( i = size - 1; i >= 1; i-- )); do
    for (( s = 1; s <= size - i; s++ )); do
        echo -n " "
    done
    for (( j = 1; j <= 2*i - 1; j++ )); do
        echo -n "*"
    done
    echo
done
