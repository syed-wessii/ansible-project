#!/bin/bash

# Assignment 7C - Git Tag Management Utility

usage() {
    echo "Usage:"
    echo "./assignment7c.sh -t <tag_name>    # Create tag"
    echo "./assignment7c.sh -l              # List tags"
    echo "./assignment7c.sh -d <tag_name>    # Delete tag"
    exit 1
}

# Check Git repository
if ! git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
    echo "[ERROR] Not a Git repository."
    exit 1
fi

case "$1" in

    # Create tag
    -t)
        if [ -z "$2" ]; then
            echo "[ERROR] Tag name required."
            usage
        fi

        if git rev-parse "$2" >/dev/null 2>&1; then
            echo "[ERROR] Tag '$2' already exists."
            exit 1
        fi

        git tag "$2"
        echo "[SUCCESS] Tag '$2' created."
        ;;

    # List tags
    -l)
        echo "Tags:"
        git tag
        ;;

    # Delete tag
    -d)
        if [ -z "$2" ]; then
            echo "[ERROR] Tag name required."
            usage
        fi

        if ! git rev-parse "refs/tags/$2" >/dev/null 2>&1; then
            echo "[ERROR] Tag '$2' does not exist."
            exit 1
        fi

        git tag -d "$2"
        echo "[SUCCESS] Tag '$2' deleted."
        ;;

    *)
        usage
        ;;

esac
