#!/bin/bash

# Assignment 7B - Git Branch Management Utility

usage() {
    echo "Usage:"
    echo "./assignment7b.sh -l"
    echo "./assignment7b.sh -b <branch_name>"
    echo "./assignment7b.sh -d <branch_name>"
    echo "./assignment7b.sh -m -1 <branch1> -2 <branch2>"
    echo "./assignment7b.sh -r -1 <branch1> -2 <branch2>"
    exit 1
}

# Check if inside Git repository
if ! git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
    echo "[ERROR] Not a Git repository."
    exit 1
fi

case "$1" in

    # List branches
    -l)
        echo "Branches:"
        git branch
        ;;

    # Create branch
    -b)
        if [ -z "$2" ]; then
            echo "[ERROR] Branch name required."
            usage
        fi

        git branch "$2"
        echo "[SUCCESS] Branch '$2' created."
        ;;

    # Delete branch
    -d)
        if [ -z "$2" ]; then
            echo "[ERROR] Branch name required."
            usage
        fi

        git branch -d "$2"
        echo "[SUCCESS] Branch '$2' deleted."
        ;;

    # Merge branch1 into branch2
    -m)
        if [ "$2" != "-1" ] || [ "$4" != "-2" ] || [ -z "$3" ] || [ -z "$5" ]; then
            usage
        fi

        branch1="$3"
        branch2="$5"

        echo "Switching to '$branch2'..."
        git checkout "$branch2"

        echo "Merging '$branch1' into '$branch2'..."
        git merge "$branch1"

        echo "[SUCCESS] '$branch1' merged into '$branch2'."
        ;;

    # Rebase branch1 onto branch2
    -r)
        if [ "$2" != "-1" ] || [ "$4" != "-2" ] || [ -z "$3" ] || [ -z "$5" ]; then
            usage
        fi

        branch1="$3"
        branch2="$5"

        echo "Switching to '$branch1'..."
        git checkout "$branch1"

        echo "Rebasing '$branch1' onto '$branch2'..."
        git rebase "$branch2"

        echo "[SUCCESS] '$branch1' rebased onto '$branch2'."
        ;;

    *)
        usage
        ;;

esac
