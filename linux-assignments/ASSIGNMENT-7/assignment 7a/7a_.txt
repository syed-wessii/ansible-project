#!/bin/bash

# Assignment 7A - Git Branching, Merge and Conflict Resolution

# Stop script if any command fails
set -e


# Make sure we are inside a Git repository
if ! git rev-parse --is-inside-work-tree > /dev/null 2>&1; then
    echo "[ERROR] This is not a Git repository."
    exit 1
fi

# Create ninja folder
echo "[1] Creating ninja folder..."
mkdir -p ninja

# Create README.md with initial content
echo "Trying fast forward merge" > ninja/README.md

# Create ninja branch and switch to it
echo "[2] Creating ninja branch..."
git checkout -b ninja

# Show status
echo "[3] Git status:"
git status

# Add and commit changes
echo "[4] Committing changes to ninja branch..."
git add ninja/README.md
git commit -m "Add README in ninja folder"

# Switch to master/main branch
echo "[5] Switching to master branch..."

if git show-ref --verify --quiet refs/heads/master; then
    git checkout master
else
    git checkout main
fi

# Merge ninja branch
echo "[6] Merging ninja branch into master..."
git merge ninja

# Modify README on master
echo "[7] Modifying README in master..."
echo "Changes in master branch" > ninja/README.md
git add ninja/README.md
git commit -m "Update README in master branch"

# Switch back to ninja
echo "[8] Switching to ninja branch..."
git checkout ninja

# Modify README on ninja
echo "[9] Modifying README in ninja..."
echo "Changes in ninja branch" > ninja/README.md
git add ninja/README.md
git commit -m "Update README in ninja branch"

# Switch to master
echo "[10] Switching back to master..."
if git show-ref --verify --quiet refs/heads/master; then
    git checkout master
else
    git checkout main
fi

# Attempt merge and intentionally create conflict
echo "[11] Creating merge conflict..."
git merge ninja || true

echo ""
echo "===== MERGE CONFLICT CREATED ====="
echo "Resolving conflict using THEIRS..."
echo ""

# THEIRS = ninja branch changes
git checkout --theirs ninja/README.md
git add ninja/README.md

# Commit the conflict resolution
git commit -m "Resolve merge conflict using theirs"

echo ""
echo "===== FINAL STATUS ====="
git status

echo ""
echo "===== FINAL README CONTENT ====="
cat ninja/README.md

echo ""
echo "===== GIT LOG ====="
git log --oneline --graph --all --decorate -10

echo ""
echo "===== Assignment 7A Completed ====="
