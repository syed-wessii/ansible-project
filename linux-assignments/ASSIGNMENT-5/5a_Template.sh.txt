#!/bin/bash

# 1. Grab the template file (the first argument)
template_file=$1

# Safety check: Ensure the file actually exists
if [ ! -f "$template_file" ]; then
    echo "[ERROR] Template file '$template_file' not found!"
    exit 1
fi

# Read the exact contents of the template file into a variable
output=$(cat "$template_file")

# Shift pushes the first argument ($1) away, so "$@" now only contains the key=value pairs
shift

# 2. Loop through all the remaining key=value arguments
for arg in "$@"; do
    # Split the argument into key and value
    key="${arg%%=*}"   # Grabs everything before the '='
    value="${arg#*=}"  # Grabs everything after the '='

    # 3. Use 'sed' to search for {{key}} and replace it with the value
    # The 'g' at the end ensures it replaces all occurrences on a line
    output=$(echo "$output" | sed "s/{{${key}}}/${value}/g")
done

# Output the final, modified text
echo "$output"
